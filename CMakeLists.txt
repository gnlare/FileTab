cmake_minimum_required(VERSION 3.20)
project(FileTab LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

option(FILETAB_BUILD_TESTS "플랫폼 독립 코어 테스트를 빌드한다" ON)

# ---------- 플랫폼 독립 코어 ----------
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(filetab_core STATIC ${CORE_SOURCES})
target_include_directories(filetab_core PUBLIC src)
if(MSVC)
  target_compile_options(filetab_core PRIVATE /utf-8 /W4 /permissive- /EHsc)
else()
  target_compile_options(filetab_core PRIVATE -Wall -Wextra)
endif()
if(WIN32)
  target_compile_definitions(filetab_core PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(filetab_core PUBLIC shlwapi user32)
endif()

# ---------- Windows 앱 ----------
if(WIN32)
  enable_language(RC)
  file(GLOB WIN_SOURCES CONFIGURE_DEPENDS src/win/*.cpp)
  add_executable(FileTab WIN32 ${WIN_SOURCES} resources/FileTab.rc)
  target_include_directories(FileTab PRIVATE resources)
  target_compile_definitions(FileTab PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(FileTab PRIVATE filetab_core
    comctl32 shlwapi shell32 ole32 oleaut32 uuid msimg32 user32 gdi32 advapi32 comdlg32 dwmapi uxtheme)
  if(MSVC)
    target_compile_options(FileTab PRIVATE /utf-8 /W4 /permissive- /EHsc /MP)
    # 매니페스트는 resources/app.manifest 를 .rc 로 포함한다(PerMonitorV2, 긴 경로, 공용 컨트롤 v6)
    target_link_options(FileTab PRIVATE /MANIFEST:NO)
    set_property(TARGET FileTab PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(FileTab PRIVATE -Wall -Wextra -municode)
    target_link_options(FileTab PRIVATE -municode -mwindows -static)
  endif()
endif()

if(FILETAB_BUILD_TESTS)
  enable_testing()
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE filetab_core)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8 /W4 /permissive- /EHsc)
  endif()
  add_test(NAME core_tests COMMAND core_tests)
endif()
